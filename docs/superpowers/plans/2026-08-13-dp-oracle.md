### Task 1: prefix 결정성 프로브 — 표집 방식을 고른다

이 태스크는 코드를 거의 안 남기고 **측정 결과로 이후 태스크의 경로를 고른다.** spec §4 는 세 경로를 열어 뒀다. 무엇을 쓸지는 추측이 아니라 이 측정이 정한다.

배경: `gen_oracle_mc.jl` 은 `CB.schedule_ood_at_closed!(c, act)` 로 **닫힌 노드 수 c 에 도달하면** 사건을 주입한다(`gen_oracle_mc.jl:284`). 그래서 필요한 결정성은 "런 전체가 재현되는가"가 **아니라** "주입 시점의 상태가 재현되는가" 뿐이다. 2026-08-11 에 관측된 발산(`n_closed 149→123`)은 런 전체 숫자이므로 이 질문에 답하지 않는다.

```julia
using JSON3
include(joinpath(@__DIR__, "..", "oracle", "gen_oracle_mc.jl"))   # 상수·CB·주입기 재사용

const N_REPEAT = 10
const AT_CLOSED = 30            # prog_b 중간 구간을 대표하는 주입점

"주입 순간의 상태 지문. 값이 아니라 **재현되는가**만 본다."
function prefix_fingerprint()
    fp = Ref{Any}(nothing)
    act = function (env)
        fp[] === nothing || return nothing
        fp[] = Dict(
            "closed"        => CB.n_closed(env),
            "sim_seconds"   => CB.sim_seconds(env),
            "n_active"      => CB.n_active_agents(env),
            "soc_sorted"    => sort(round.(CB.agent_socs(env); digits = 6)),
            "pending_total" => CB.total_pending(env),
        )
        return nothing            # 주입하지 않는다. 지문만 찍는다.
    end
    CB.schedule_ood_at_closed!(AT_CLOSED, act)
    run_one((env, ev) -> nothing; inject = false, hz_seed = nothing)
    return fp[]
end

fps = [prefix_fingerprint() for _ in 1:N_REPEAT]
identical = all(f -> f == fps[1], fps)
println(JSON3.write(Dict("identical" => identical, "n" => N_REPEAT, "fingerprints" => fps)))
```

```python
"""dp_grid.py -- DP oracle 격자의 파이썬 쪽 구현. 정의 자체는 grid_spec.json 에만 있다.

왜 spec 을 JSON 으로 빼는가: 격자 정의가 코드 두 곳(파이썬 채점기 / Julia 표집기)에
복붙되면 조용히 갈린다. 이 저장소는 그 사고를 이미 겪었다(action_registry.json 머리말).
"""
import json
import math
import os

HERE = os.path.dirname(os.path.abspath(__file__))
SPEC_PATH = os.path.join(HERE, "dp_oracle", "grid_spec.json")

_TRUTH_TO_EVT = {"BatteryTruth": "Battery", "FaultTruth": "Fault",
                 "ZoneTruth": "Zone", "ReformTruth": "Reform"}


def load_spec(path=None):
    with open(path or SPEC_PATH, encoding="utf-8") as fh:
        return json.load(fh)


_SPEC = None


def _spec():
    global _SPEC
    if _SPEC is None:
        _SPEC = load_spec()
    return _SPEC


def _finite(x):
    """None/파싱불가/NaN/Inf 를 전부 None 으로. float('nan') <= 0.02 는 조용히 False 라서
    이 게이트 없이 비교하면 NaN 이 엉뚱한 칸으로 들어간다 (reference_policy._finite_soc 와 같은 이유)."""
    try:
        v = float(x)
    except (TypeError, ValueError):
        return None
    return v if math.isfinite(v) else None


def _bin(v, edges):
    """edges = [e0, e1, ..., en]. 구간 i = (e_i-1, e_i], 단 최저 구간은 [e0, e1] 로 닫는다.
    경계값이 아래 칸에 들어가야 soc 0.02 가 '완주로 갈린 칸' 에 남는다."""
    for i in range(1, len(edges)):
        if v <= edges[i] if i == 1 else (edges[i - 1] < v <= edges[i]):
            return i - 1
    return len(edges) - 2


def phi(ev):
    """결정 사건 dict -> 격자칸 dict. 필수 필드가 없으면 None (unscored)."""
    sp = _spec()["axes"]
    evt = _TRUTH_TO_EVT.get(str(ev.get("truth", "")))
    if evt is None:
        return None

    prog = _finite(ev.get("progress"))
    spare = _finite(ev.get("spare_count"))
    if prog is None or spare is None:
        return None

    soc = _finite(ev.get("soc"))
    if evt == "Battery" and soc is None:
        return None            # 배터리 결정인데 SoC 가 없으면 채점할 수 없다
    if soc is None:
        soc = 1.0              # 배터리 사건이 아니면 SoC 는 상한 칸으로 (정보 없음 아님, '문제 아님')

    zp = ev.get("zone_primitives") or {}
    nb, rc = zp.get("n_nav_blocked"), zp.get("root_covered")
    if evt == "Zone" and (nb is None or rc is None):
        return None
    if nb is None or rc is None:
        zone_s = "none"
    elif int(rc) > 0:
        zone_s = "cov"
    elif int(nb) > 0:
        zone_s = "blk"
    else:
        zone_s = "none"

    pend = ev.get("pend_f")
    if pend is None:
        # 런 요약에는 fleet 단위 pending fault 수가 없다. 이 사건이 fault 면 최소 1건이다.
        pend = 1 if evt == "Fault" else 0
    pend = min(int(pend), sp["pend_f"]["cap"])

    return dict(
        prog_b=_bin(prog, sp["prog_b"]["edges"]),
        soc_b=_bin(soc, sp["soc_b"]["edges"]),
        spares_b=_bin(spare, sp["spares_b"]["edges"]),
        pend_f=pend,
        zone_s=zone_s,
        evt=evt,
    )


def node_key(cell):
    return "p%d|s%d|sp%d|f%d|%s|%s" % (cell["prog_b"], cell["soc_b"], cell["spares_b"],
                                       cell["pend_f"], cell["zone_s"], cell["evt"])


def is_feasible(cell):
    for rule in _spec()["infeasible_rules"]:
        if not all(cell.get(k) == v for k, v in rule["when"].items()):
            continue
        for key, want in rule["require"].items():
            axis, _, op = key.partition("__")
            got = cell.get(axis)
            if op == "max" and got > want:
                return False
            if op == "min" and got < want:
                return False
            if op == "not" and got == want:
                return False
    return True


def all_cells():
    sp = _spec()["axes"]
    out = []
    for p in range(len(sp["prog_b"]["edges"]) - 1):
        for s in range(len(sp["soc_b"]["edges"]) - 1):
            for spb in range(len(sp["spares_b"]["edges"]) - 1):
                for f in range(sp["pend_f"]["cap"] + 1):
                    for z in sp["zone_s"]["values"]:
                        for e in sp["evt"]["values"]:
                            c = dict(prog_b=p, soc_b=s, spares_b=spb, pend_f=f, zone_s=z, evt=e)
                            if is_feasible(c):
                                out.append(c)
    return out
```

```python
    rows = []
    with open(path, encoding="utf-8") as fh:

            line = line.strip()
```

```python
"""dp_solve.py -- samples.jsonl -> value.json. 순수 파이썬, 시뮬 호출 없음, 결정적.

구조 (spec §7):
  버킷 사이는 prog_b 역순 backward induction (progress 단조성이 DAG 를 보장),
  버킷 내부는 자기순환이 있으므로 value iteration 으로 수렴시킨다.
  비용 최소화(SSP)이므로 min 이다 -- 보상 최대화가 아니다.
"""
import collections
import json
import math

MAX_ITERS = 200
TOL = 1e-6


def _load(path):
    rows = []
    with open(path, encoding="utf-8") as fh:
        for line in fh:
            line = line.strip()
            if line:
                rows.append(json.loads(line))
    return rows


def _prog_of(node_key):
    """node_key 'p2|...' 에서 prog_b 를 뽑는다. 버킷 순서를 정하는 데만 쓴다."""
    return int(node_key.split("|", 1)[0][1:])


def solve(samples_path, spec):
    rows = _load(samples_path)
    c_fail = float(spec.get("C_fail", 10000.0))
    tie_z = float(spec.get("tie_z", 1.96))

    # (node, arm_name) -> [(cost, next_node, next_terminal), ...]
    trans = collections.defaultdict(list)
    for r in rows:
        trans[(r["node"], r["arm_name"])].append(
            (float(r["cost"]), r.get("next_node"), r.get("next_terminal")))

    nodes = sorted({n for n, _ in trans})
    V = {n: c_fail for n in nodes}          # 낙관 초기화 금지: 위에서 내려온다
    converged = {n: False for n in nodes}

    def q(node, arm):
        vals = []
        for cost, nxt, term in trans[(node, arm)]:
            if term == "goal":
                vals.append(cost)
            elif term == "dead_end":
                vals.append(cost + c_fail)
            else:
                vals.append(cost + V.get(nxt, c_fail))
        return vals

    # prog_b 역순으로 버킷을 돌고, 각 버킷 안에서 수렴할 때까지 반복한다.
    iters_used = 0
    for p in sorted({_prog_of(n) for n in nodes}, reverse=True):
        bucket = [n for n in nodes if _prog_of(n) == p]
        for it in range(MAX_ITERS):
            delta = 0.0
            for n in bucket:
                arms = sorted({a for (nn, a) in trans if nn == n})
                if not arms:
                    continue
                new = min(sum(q(n, a)) / len(q(n, a)) for a in arms)
                delta = max(delta, abs(new - V[n]))
                V[n] = new
            iters_used = max(iters_used, it + 1)
            if delta < TOL:
                for n in bucket:
                    converged[n] = True
                break

    out_nodes = {}
    for n in nodes:
        arms = sorted({a for (nn, a) in trans if nn == n})
        qs = {}
        for a in arms:
            vals = q(n, a)
            mean = sum(vals) / len(vals)
            sd = math.sqrt(sum((v - mean) ** 2 for v in vals) / (len(vals) - 1)) if len(vals) > 1 else 0.0
            qs[a] = dict(mean=mean, se=sd / math.sqrt(len(vals)) if vals else 0.0, n=len(vals))
        best = min(qs, key=lambda a: qs[a]["mean"])
        band = tie_z * math.sqrt(qs[best]["se"] ** 2 + max(
            (qs[a]["se"] ** 2 for a in qs if a != best), default=0.0))
        tied = sorted(a for a in qs if qs[a]["mean"] - qs[best]["mean"] <= band)
        out_nodes[n] = dict(V=V[n], a_star=tied, tie=len(tied) > 1,
                            converged=converged[n], Q=qs)

    return dict(nodes=out_nodes,
                meta=dict(C_fail=c_fail, tie_z=tie_z, iters=iters_used,
                          n_capped=sum(1 for r in rows if r.get("capped")),
                          n_rows=len(rows)))


def a_star(value, node_key):
    node = value["nodes"].get(node_key)
    return node["a_star"] if node else None
```

```python
"""dp_oracle_policy.py -- DP 값표에서 기준행동 a* 를 조회한다.

reference_policy.py 를 대체한다. 시그니처가 같아서 소비처는 import 만 바꾸면 된다.

reference_policy.py 와 다른 점 셋:
  1) a* 가 규칙이 아니라 **격자 위 backward induction 의 argmin** 이다.
  2) 조합 사건(pending fault + zone + 낮은 SoC 가 동시)도 상태로 구별되므로 a* 가 나온다.
  3) 동점을 동점으로 본다 -- tie 집합 안의 선택은 정답으로 센다(MDP_DESIGN §7.1).
없는 답을 지어내지 않는다는 계약은 그대로다: 표에 없는 칸은 None(unscored).
"""
import json
import os
import sys

HERE = os.path.dirname(os.path.abspath(__file__))
if HERE not in sys.path:
    sys.path.insert(0, HERE)

import dp_grid

VALUE_PATH = os.path.join(HERE, "dp_oracle", "value.json")
_VALUE = None


def _value():
    global _VALUE
    if _VALUE is None:
        with open(VALUE_PATH, encoding="utf-8") as fh:
            _VALUE = json.load(fh)
    return _VALUE


def reference_actions(ev):
    """tie 집합 전체를 돌려준다. 표에 없거나 상태가 불완전하면 None."""
    cell = dp_grid.phi(ev)
    if cell is None:
        return None
    node = _value()["nodes"].get(dp_grid.node_key(cell))
    return node["a_star"] if node else None


def reference_action(ev):
    """(a_star, basis, note). reference_policy.reference_action 과 같은 계약."""
    cell = dp_grid.phi(ev)
    if cell is None:
        return None, "dp", "상태 필드가 불완전해 격자칸을 정할 수 없다 -- unscored"
    key = dp_grid.node_key(cell)
    node = _value()["nodes"].get(key)
    if node is None:
        return None, "dp", "격자 표에 없는 칸(%s) -- 표집되지 않았으므로 unscored" % key
    arms = node["a_star"]
    note = "DP V=%.1f, %s" % (node["V"], "tie %s" % arms if node["tie"] else "단일 argmin")
    # tie 면 첫 팔을 대표로 돌려주되, 채점은 score() 가 집합으로 한다.
    return arms[0], "dp", note


def score(decisions):
    """결정 목록을 채점해 (scored, correct, rows) 를 돌려준다.
    tie 집합 안의 선택은 정답이다 -- 격차가 노이즈보다 작은 것을 오답으로 세면
    표집 노이즈를 정책의 실수로 오인하게 된다."""
    rows = []
    for ev in decisions:
        arms = reference_actions(ev)
        a_star, basis, note = reference_action(ev)
        chosen = str(ev.get("macro") or "")
        rows.append(dict(truth=ev.get("truth"), at=ev.get("at"), chosen=chosen,
                         reference=a_star, reference_set=arms, basis=basis, note=note,
                         correct=(None if arms is None else chosen in arms)))
    scored = [r for r in rows if r["correct"] is not None]
    return len(scored), sum(1 for r in scored if r["correct"]), rows
```
